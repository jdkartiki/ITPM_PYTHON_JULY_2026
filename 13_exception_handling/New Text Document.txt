# ---------------- CUSTOM EXCEPTIONS ----------------

class ProductNotFoundException(Exception):
    pass


class InvalidQuantityException(Exception):
    pass


class InsufficientStockException(Exception):
    pass


class EmptyCartException(Exception):
    pass


# ---------------- PRODUCT CLASS ----------------

class Product:

    def __init__(self, product_id, product_name, price, quantity_available):

        self.product_id = product_id
        self.product_name = product_name
        self.__price = price
        self.__quantity_available = quantity_available

    # Display product
    def display_product(self):

        print(
            self.product_id,
            "|",
            self.product_name,
            "| ₹",
            self.__price,
            "| Stock:",
            self.__quantity_available
        )

    # Get price
    def get_price(self):

        return self.__price

    # Get stock
    def get_stock(self):

        return self.__quantity_available

    # Update stock
    def update_stock(self, quantity):

        self.__quantity_available += quantity


# ---------------- SHOPPING CART CLASS ----------------

class ShoppingCart:

    def __init__(self):

        self.cart_items = {}

    # Add product
    def add_product(self, product, quantity):

        if quantity <= 0:
            raise InvalidQuantityException(
                "Quantity must be greater than 0."
            )

        if quantity > product.get_stock():
            raise InsufficientStockException(
                "Only "
                + str(product.get_stock())
                + " units are available."
            )

        # If product already exists
        if product.product_id in self.cart_items:

            self.cart_items[product.product_id]["quantity"] += quantity

        else:

            self.cart_items[product.product_id] = {
                "product": product,
                "quantity": quantity
            }

        # Reduce product stock
        product.update_stock(-quantity)

        print("\nProduct added to cart successfully!")

    # Remove product
    def remove_product(self, product_id):

        if product_id not in self.cart_items:

            raise ProductNotFoundException(
                "Product is not present in cart."
            )

        item = self.cart_items[product_id]

        product = item["product"]
        quantity = item["quantity"]

        # Return stock
        product.update_stock(quantity)

        del self.cart_items[product_id]

        print("\nProduct removed from cart successfully!")

    # Update quantity
    def update_quantity(self, product_id, new_quantity):

        if product_id not in self.cart_items:

            raise ProductNotFoundException(
                "Product is not present in cart."
            )

        if new_quantity <= 0:

            raise InvalidQuantityException(
                "Quantity must be greater than 0."
            )

        item = self.cart_items[product_id]

        product = item["product"]
        old_quantity = item["quantity"]

        difference = new_quantity - old_quantity

        # Increasing quantity
        if difference > 0:

            if difference > product.get_stock():

                raise InsufficientStockException(
                    "Not enough stock available."
                )

            product.update_stock(-difference)

        # Decreasing quantity
        elif difference < 0:

            product.update_stock(-difference)

        item["quantity"] = new_quantity

        print("\nCart quantity updated successfully!")

    # Calculate total
    def calculate_total(self):

        total = 0

        for item in self.cart_items.values():

            product = item["product"]
            quantity = item["quantity"]

            total += product.get_price() * quantity

        return total

    # Display cart
    def display_cart(self):

        if len(self.cart_items) == 0:

            raise EmptyCartException(
                "Cart is empty."
            )

        print("\n========== SHOPPING CART ==========")

        for item in self.cart_items.values():

            product = item["product"]
            quantity = item["quantity"]

            price = product.get_price()

            subtotal = price * quantity

            print("\nProduct ID ::", product.product_id)
            print("Product    ::", product.product_name)
            print("Price      :: ₹", price)
            print("Quantity   ::", quantity)
            print("Subtotal   :: ₹", subtotal)

        print("\n===================================")
        print("Total Amount :: ₹", self.calculate_total())

    # Checkout
    def checkout(self):

        if len(self.cart_items) == 0:

            raise EmptyCartException(
                "Cannot checkout. Cart is empty."
            )

        print("\n========== FINAL BILL ==========")

        for item in self.cart_items.values():

            product = item["product"]
            quantity = item["quantity"]

            price = product.get_price()
            subtotal = price * quantity

            print(
                product.product_name,
                "x",
                quantity,
                " = ₹",
                subtotal
            )

        total = self.calculate_total()

        print("--------------------------------")
        print("Total Amount :: ₹", total)
        print("Order placed successfully!")
        print("Thank you for shopping with us!")

        # Empty cart after checkout
        self.cart_items.clear()


# ---------------- FIND PRODUCT ----------------

def find_product(products, product_id):

    for product in products:

        if product.product_id == product_id:

            return product

    raise ProductNotFoundException(
        "Product ID does not exist."
    )

# ---------------- CUSTOM EXCEPTIONS ----------------

class InsufficientBalanceException(Exception):
    pass


class InvalidAccountException(Exception):
    pass


class InvalidAmountException(Exception):
    pass


# ---------------- BANK ACCOUNT CLASS ----------------

class BankAccount:

    def __init__(self, account_number, customer_name, account_type, balance):
        self.account_number = account_number
        self.customer_name = customer_name
        self.account_type = account_type
        self.__balance = balance

    # Display account details
    def display_account_details(self):
        print("\n========== ACCOUNT DETAILS ==========")
        print("Account Number ::", self.account_number)
        print("Customer Name  ::", self.customer_name)
        print("Account Type   ::", self.account_type)
        print("Balance        :: ₹", self.__balance)

    # Deposit money
    def deposit(self, amount):

        if amount <= 0:
            raise InvalidAmountException(
                "Deposit amount must be greater than 0."
            )

        self.__balance += amount

        print("\nDeposit Successful!")
        print("Deposited Amount :: ₹", amount)
        print("Current Balance  :: ₹", self.__balance)

    # Withdraw money
    def withdraw(self, amount):

        if amount <= 0:
            raise InvalidAmountException(
                "Withdrawal amount must be greater than 0."
            )

        if amount > self.__balance:
            raise InsufficientBalanceException(
                "Insufficient Balance."
            )

        self.__balance -= amount

        print("\nWithdrawal Successful!")
        print("Withdrawn Amount :: ₹", amount)
        print("Current Balance  :: ₹", self.__balance)

    # Check balance
    def check_balance(self):

        print("\nCurrent Balance :: ₹", self.__balance)

    # Transfer money
    def transfer_amount(self, amount, target_account):

        if amount <= 0:
            raise InvalidAmountException(
                "Transfer amount must be greater than 0."
            )

        if amount > self.__balance:
            raise InsufficientBalanceException(
                "Insufficient Balance for transfer."
            )

        self.__balance -= amount
        target_account.__balance += amount

        print("\nTransfer Successful!")
        print("Transferred Amount :: ₹", amount)
        print("From Account      ::", self.account_number)
        print("To Account        ::", target_account.account_number)
        print("Remaining Balance  :: ₹", self.__balance)


# ---------------- FIND ACCOUNT FUNCTION ----------------

def find_account(accounts, account_number):

    for account in accounts:

        if account.account_number == account_number:
            return account

    raise InvalidAccountException(
        "Account number does not exist."
    )
